    <script>
        // --- System Settings ---
        const apiKey = ""; // המפתח מסופק אוטומטית על ידי סביבת ההרצה

        // --- Local Storage AI Settings ---
        function loadAISettings() {
            const savedKey = localStorage.getItem('cashflow_api_key') || '';
            const savedModel = localStorage.getItem('cashflow_ai_model') || 'gemini-2.5-flash';
            
            const keyInput = document.getElementById('userApiKey');
            const modelSelect = document.getElementById('userAiModel');
            
            if (keyInput) keyInput.value = savedKey;
            if (modelSelect) modelSelect.value = savedModel;
        }

        function saveAISettings() {
            const key = document.getElementById('userApiKey').value.trim();
            const model = document.getElementById('userAiModel').value;
            localStorage.setItem('cashflow_api_key', key);
            localStorage.setItem('cashflow_ai_model', model);
        }

        // --- Database & Persistence setup (IndexedDB) ---
        const DB_NAME = "CashflowDB";
        let db;

        // Default initial state
        function getInitialState() {
            return {
                cash: 1000,
                incomes: [{ id: 1, name: 'משכורת', amount: 3000, isPassive: false }],
                expenses: [
                    { id: 1, name: 'מסים', amount: 500 },
                    { id: 2, name: 'משכנתא', amount: 400 },
                    { id: 3, name: 'הלוואות ואשראי', amount: 300 },
                    { id: 4, name: 'הוצאות מחיה', amount: 1200 }
                ],
                assets: [],
                liabilities: [],
                history: []
            };
        }

        let state = getInitialState();

        // Initialize Database
        const request = indexedDB.open(DB_NAME, 1);
        request.onerror = (e) => {
            console.error("Database error", e);
            updateUI(); // Fallback to memory
        };
        request.onupgradeneeded = (e) => {
            db = e.target.result;
            db.createObjectStore("stateStore", { keyPath: "id" });
        };
        request.onsuccess = (e) => {
            db = e.target.result;
            loadState();
        };

        function saveState() {
            if (!db) return;
            try {
                const transaction = db.transaction(["stateStore"], "readwrite");
                const store = transaction.objectStore("stateStore");
                store.put({ id: "currentState", data: state });
            } catch (error) {
                console.error("Save error", error);
            }
        }

        function loadState() {
            if (!db) return;
            const transaction = db.transaction(["stateStore"], "readonly");
            const store = transaction.objectStore("stateStore");
            const req = store.get("currentState");
            req.onsuccess = (e) => {
                if (req.result && req.result.data) {
                    state = req.result.data;
                    // Ensure history exists for players migrating from older versions
                    if (!state.history) state.history = [];
                }
                updateUI();
            };
            req.onerror = () => updateUI(); // Render defaults on fail
        }

        function resetGame() {
            if (confirm("האם אתה בטוח שברצונך לאפס את המשחק לגמרי? כל הנתונים וההיסטוריה יימחקו לצמיתות.")) {
                state = getInitialState();
                updateUI();
            }
        }

        // --- Core Functions ---
        function updateUI() {
            // Calculations
            const totalIncome = state.incomes.reduce((sum, item) => sum + item.amount, 0);
            const passiveIncome = state.incomes.filter(i => i.isPassive).reduce((sum, item) => sum + item.amount, 0);
            const totalExpenses = state.expenses.reduce((sum, item) => sum + item.amount, 0);
            const payday = totalIncome - totalExpenses;
            const isWinner = passiveIncome >= totalExpenses && totalExpenses > 0;
            const progressPercent = totalExpenses > 0 ? Math.min((passiveIncome / totalExpenses) * 100, 100) : 0;

            // Header Updates
            document.getElementById('cashVal').innerText = state.cash.toLocaleString();
            document.getElementById('paydayVal').innerText = payday.toLocaleString();
            document.getElementById('passiveVal').innerText = passiveIncome.toLocaleString();
            document.getElementById('expensesTotalVal').innerText = totalExpenses.toLocaleString();
            
            const paydayContainer = document.getElementById('paydayContainer');
            paydayContainer.className = `text-xl font-bold ${payday >= 0 ? 'text-emerald-400' : 'text-red-400'}`;

            // Progress Bar
            const pBar = document.getElementById('progressBar');
            pBar.style.width = `${progressPercent}%`;
            pBar.className = `h-full transition-all duration-500 ease-in-out ${isWinner ? 'bg-amber-400' : 'bg-emerald-500'}`;
            document.getElementById('progressText').innerText = `${Math.round(progressPercent)}% לעצמאות כלכלית`;
            
            if(isWinner) document.getElementById('winnerBadge').classList.remove('hidden');
            else document.getElementById('winnerBadge').classList.add('hidden');

            // Dashboard Updates
            document.getElementById('dashIncome').innerText = totalIncome.toLocaleString();
            document.getElementById('dashExpense').innerText = totalExpenses.toLocaleString();

            // Render History
            renderHistory();

            // Render Lists
            renderList('incomesList', state.incomes, 'incomes', 'fa-wallet', 'bg-emerald-100 text-emerald-600');
            renderList('expensesList', state.expenses, 'expenses', 'fa-circle-exclamation', 'bg-red-100 text-red-600');
            renderList('assetsList', state.assets, 'assets', 'fa-building', 'bg-blue-100 text-blue-600');
            renderList('liabilitiesList', state.liabilities, 'liabilities', 'fa-circle-exclamation', 'bg-orange-100 text-orange-600');

            // Save changes to DB
            saveState();
        }

        function renderHistory() {
            const container = document.getElementById('historyList');
            if (state.history && state.history.length > 0) {
                container.innerHTML = state.history.map(item => `
                    <div class="flex justify-between items-center text-sm p-3 bg-slate-50 rounded-xl border border-slate-100">
                        <div class="flex flex-col">
                            <span class="font-medium text-slate-800">${item.desc}</span>
                            <span class="text-[10px] text-slate-400 mt-0.5"><i class="fa-regular fa-clock mr-1"></i>${item.time}</span>
                        </div>
                        <span class="font-black ${item.amount > 0 ? 'text-emerald-500' : 'text-red-500'} bg-white px-3 py-1 rounded-lg border border-slate-200">
                            ${item.amount > 0 ? '+' : ''}${item.amount.toLocaleString()}$
                        </span>
                    </div>
                `).join('');
            } else {
                container.innerHTML = '<p class="text-xs text-center text-slate-400 py-3">אין פעולות מהירות עדיין.</p>';
            }
        }

        function renderList(containerId, items, listName, iconClass, colorClass) {
            const container = document.getElementById(containerId);
            if(items.length === 0 && containerId === 'assetsList') {
                container.innerHTML = '<p class="text-sm text-slate-400 text-center py-4">אין נכסים כרגע.</p>';
                return;
            }
            container.innerHTML = items.map(item => `
                <div class="flex justify-between items-center p-3 mb-2 bg-white rounded-xl shadow-sm border border-slate-100">
                    <div class="flex items-center gap-3">
                        <div class="p-2 rounded-lg ${colorClass}">
                            <i class="fa-solid ${iconClass}"></i>
                        </div>
                        <div>
                            <p class="font-semibold text-slate-800 text-sm">${item.name}</p>
                            ${item.isPassive !== undefined ? `<p class="text-xs text-slate-500">${item.isPassive ? 'פסיבי' : 'משכורת'}</p>` : ''}
                        </div>
                    </div>
                    <div class="flex items-center gap-4">
                        <span class="font-bold text-slate-800">$${item.amount.toLocaleString()}</span>
                        <button onclick="removeItem('${listName}', ${item.id})" class="text-red-400 hover:text-red-600 p-1">
                            <i class="fa-solid fa-trash"></i>
                        </button>
                    </div>
                </div>
            `).join('');
        }

        // --- Actions ---
        function addCash(amount, description = 'פעולה ללא שם') {
            const now = new Date();
            const timeStr = now.getHours().toString().padStart(2, '0') + ':' + now.getMinutes().toString().padStart(2, '0');
            
            // Push to history
            state.history.unshift({
                id: Date.now(),
                desc: description,
                amount: amount,
                time: timeStr
            });
            
            // Keep history length manageable (last 30 items)
            if (state.history.length > 30) state.history.pop();
            
            state.cash += amount;
            updateUI();
        }

        function handlePayday() {
            const totalIncome = state.incomes.reduce((sum, item) => sum + item.amount, 0);
            const totalExpenses = state.expenses.reduce((sum, item) => sum + item.amount, 0);
            const netPayday = totalIncome - totalExpenses;
            
            addCash(netPayday, 'קבלת משכורת חודשית (תזרים מזומנים)');
        }

        function handleQuickCash(action) {
            const inputAmount = document.getElementById('quickCash');
            const inputDesc = document.getElementById('quickCashDesc');
            
            const val = Number(inputAmount.value);
            const desc = inputDesc.value.trim() || (action === 'add' ? 'הכנסה כללית' : 'הוצאה כללית');
            
            if (val && val > 0) {
                if(action === 'add') addCash(val, desc);
                if(action === 'sub') addCash(-val, desc);
                
                // Clear fields
                inputAmount.value = '';
                inputDesc.value = '';
            }
        }

        function addItem(e, listName) {
            e.preventDefault();
            let nameId, amountId, passiveId;
            
            if(listName === 'incomes') { nameId = 'incName'; amountId = 'incAmount'; passiveId = 'incPassive'; }
            else if(listName === 'expenses') { nameId = 'expName'; amountId = 'expAmount'; }
            else if(listName === 'assets') { nameId = 'astName'; amountId = 'astAmount'; }
            else if(listName === 'liabilities') { nameId = 'libName'; amountId = 'libAmount'; }

            const name = document.getElementById(nameId).value;
            const amount = Number(document.getElementById(amountId).value);
            
            if(!name || !amount) return;

            const newItem = { id: Date.now(), name, amount };
            if(passiveId) newItem.isPassive = document.getElementById(passiveId).checked;

            state[listName].push(newItem);
            
            // Clear inputs
            document.getElementById(nameId).value = '';
            document.getElementById(amountId).value = '';
            if(passiveId) document.getElementById(passiveId).checked = false;

            updateUI();
        }

        function removeItem(listName, id) {
            state[listName] = state[listName].filter(item => item.id !== id);
            updateUI();
        }

        // --- UI Navigation ---
        function switchTab(tabId, btn) {
            // Hide all tabs
            ['dashboard', 'flow', 'assets', 'ai'].forEach(t => {
                document.getElementById(`tab-${t}`).classList.add('hidden');
            });
            // Show selected
            document.getElementById(`tab-${tabId}`).classList.remove('hidden');

            // Reset all buttons
            document.querySelectorAll('.nav-btn').forEach(b => {
                b.className = 'nav-btn text-slate-400 hover:text-slate-600 flex flex-col items-center justify-center w-full py-2 px-1 rounded-xl transition-all';
                b.querySelector('i').classList.add('opacity-80');
            });

            // Activate clicked button
            btn.className = 'nav-btn text-blue-600 font-bold bg-blue-50 flex flex-col items-center justify-center w-full py-2 px-1 rounded-xl transition-all';
            btn.querySelector('i').classList.remove('opacity-80');
        }

                resContainer.classList.remove('hidden');

            }

            // Loading state
            btn.disabled = true;
            btn.classList.add('opacity-50');

            btnIcon.classList.add('fa-spinner', 'fa-spin');

            const totalIncome = state.incomes.reduce((s, i) => s + i.amount, 0);

            const totalExpenses = state.expenses.reduce((s, i) => s + i.amount, 0);

            // השדרוג של הפרומפט כדי שיבין מתמטית את חוקי הלוואות ומשחק קאשפלואו
            const prompt = `

חוקי ברזל חובה לחישוב:

קיבלתי את העסקה הבאה: "${input}"

על סמך כל זה (חשב בעצמך כמה הלוואה חסרה בכפולות של 1000 ומה הריבית עליה), תן לי המלצה במשפט אחד בלבד בעברית.

למשל: "כן, תיקח הלוואה של 2000$ שתעלה לך 200$ בחודש, והיות והנכס מכניס 400$, אתה תרוויח נטו 200$ בתזרים כל חודש."
